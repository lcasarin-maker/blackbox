```python
import glob
import json

MAX42 = 42 * 1024 * 1024  # kB
files = sorted(glob.glob('/home/lcasarin/.local/share/blackbox/samples/*.jsonl'))
files = [f for f in files if f.split('/')[-1] >= '2026-09-28']
rows = []
for f in files:
    with open(f, encoding='utf-8') as fh:
        for line in fh:
            try:
                d = json.loads(line)
            except ValueError:
                continue
            if d.get('burst') or 'slices' not in d:
                continue
            app = next((s for s in d.get('slices') or [] if s.get('slice') == 'app'), None)
            psi = (d.get('psi') or {}).get('mem_full')
            if app is None or psi is None:
                continue
            pid0 = (d.get('pidio') or [None])[0]
            rows.append((d['ts'], d.get('boot_id'), float(psi), app['cur_kb'] / MAX42, pid0))
print('muestras con app.slice y psi desde 2026-09-28:', len(rows), rows[0][0], '->', rows[-1][0])


def cnt(pred):
    return sum(1 for r in rows if pred(r))


for thr in (0.80, 0.85, 0.90, 0.93):
    print(f'ratio>={thr:.2f}: total={cnt(lambda r: r[3] >= thr)}  con psi>=10: '
          f'{cnt(lambda r: r[3] >= thr and r[2] >= 10)}  con psi<10: {cnt(lambda r: r[3] >= thr and r[2] < 10)}')
print('psi>=10 total:', cnt(lambda r: r[2] >= 10), ' de ellas ratio<0.90:', cnt(lambda r: r[2] >= 10 and r[3] < 0.90))
print('\nmuestras psi>=10 (ts, psi, ratio):')
for r in rows:
    if r[2] >= 10:
        print(' ', r[0], r[2], f'{r[3]*100:.1f}%')
print('\nmax ratio con psi<10:', max(r[3] for r in rows if r[2] < 10))
ratios = sorted(r[3] for r in rows)
for q in (0.5, 0.9, 0.99, 0.999):
    print(f'p{q*100:g} ratio', f'{ratios[int(q*(len(ratios)-1))]*100:.1f}%')
g = sorted(r[4]['crecio_kb'] for r in rows if r[4] and not r[4].get('nuevo'))
print('\npidio[0] no-nuevo n=', len(g))
for q in (0.5, 0.9, 0.99, 0.999, 1.0):
    print(f' p{q*100:g}', g[int(q*(len(g)-1))] // 1024, 'MiB')
print('pidio[0] en muestras con psi>=10:')
for r in rows:
    if r[2] >= 10 and r[4]:
        print(' ', r[0], r[4])
```

## Script 2 (calib2.py): disparo del corte de origen por crecimiento
```python
import glob
import json

files = sorted(glob.glob('/home/lcasarin/.local/share/blackbox/samples/*.jsonl'))
files = [f for f in files if f.split('/')[-1] >= '2026-09-28']
n = 0
hi = []
psi_hi = []
for f in files:
    with open(f, encoding='utf-8') as fh:
        for line in fh:
            try:
                d = json.loads(line)
            except ValueError:
                continue
            if d.get('burst') or 'pidio' not in d or 'slices' not in d:
                continue
            n += 1
            p = (d.get('pidio') or [None])[0]
            c = p['crecio_kb'] if p else 0
            hi.append(c)
            if float((d.get('psi') or {}).get('mem_full') or 0) >= 10:
                psi_hi.append(c)
print('muestras completas desde 2026-09-28:', n)
for gib in (1, 2, 4, 8, 16):
    kb = gib * 1024 * 1024
    a = sum(1 for c in hi if c >= kb)
    b = sum(1 for c in psi_hi if c >= kb)
    print(f'pidio[0].crecio_kb >= {gib} GiB: {a} muestras ({100*a/n:.2f} %, {a/15:.1f}/dia en 15 dias); '
          f'de las {len(psi_hi)} con PSI>=10 cubre {b}')
```
