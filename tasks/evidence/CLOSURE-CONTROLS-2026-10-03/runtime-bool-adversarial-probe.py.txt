def walk(x,path=()):
 if isinstance(x,dict):
  for k,v in x.items(): yield from walk(v,path+(k,))
 elif isinstance(x,list):
  for i,v in enumerate(x): yield from walk(v,path+(i,))
 elif type(x) is int: yield path,x
